Panduan Integrasi Frontend GitHub & Backend Apps Script

Ikuti langkah-langkah di bawah untuk menghubungkan aplikasi ini ke Google Sheets via Apps Script dan meng-host di GitHub Pages.

1 Setup Backend Google Apps Script
  1. Buka Google Sheets baru. Buat header kolom di Sheet1: id | nama | fakultas | status | timestamp
  2. Klik menu Ekstensi > Apps Script.
  3. Hapus semua kode lama dan paste kode Apps Script di bawah.
  4. Klik Terapkan (Deploy) > Terapkan sebagai Aplikasi Web (Web App).
  5. Setel Akses Siapa Saja (Anyone), lalu Klik Deploy. Copy Web App URL dan masukkan di input sync atas app ini.
2 Connect & Hosting di GitHub Pages
  1. Buat repository baru di GitHub (misal: presensi-qr-app).
  2. Simpan file ini dengan nama index.html.
  3. Push file index.html ke repository GitHub Anda.
  4. Masuk ke Settings repository > Pages.
  5. Pada bagian Branch, pilih main / root lalu simpan (Save).
  6. Aplikasi Anda kini live di URL GitHub Pages!
Script Backend Google Apps Script (Code.gs) 
function doGet(e) {
  var sheet = SpreadsheetApp.getActiveSpreadsheet().getActiveSheet();
  var data = sheet.getDataRange().getValues();
  var rows = [];
  
  for (var i = 1; i < data.length; i++) {
    rows.push({
      id: String(data[i][0]),
      nama: data[i][1],
      fakultas: data[i][2],
      status: data[i][3] || 'Belum Hadir',
      timestamp: data[i][4] || '-'
    });
  }
  
  return ContentService.createTextOutput(JSON.stringify({ status: "success", data: rows }))
    .setMimeType(ContentService.MimeType.JSON);
}

function doPost(e) {
  try {
    var contents = JSON.parse(e.postData.contents);
    var sheet = SpreadsheetApp.getActiveSpreadsheet().getActiveSheet();
    var data = sheet.getDataRange().getValues();
    
    // Update status or Sync whole list
    if (contents.action === "updateStatus") {
      for (var i = 1; i < data.length; i++) {
        if (String(data[i][0]) === String(contents.id)) {
          sheet.getRange(i + 1, 4).setValue("Hadir");
          sheet.getRange(i + 1, 5).setValue(contents.timestamp);
          break;
        }
      }
    } else if (contents.action === "syncAll") {
      sheet.clearContents();
      sheet.appendRow(["ID Peserta", "Nama Peserta", "Fakultas", "Status Kehadiran", "Waktu Presensi"]);
      contents.participants.forEach(function(p) {
        sheet.appendRow([p.id, p.nama, p.fakultas, p.status, p.timestamp || '-']);
      });
    }
    
    return ContentService.createTextOutput(JSON.stringify({ status: "success" }))
      .setMimeType(ContentService.MimeType.JSON);
  } catch (error) {
    return ContentService.createTextOutput(JSON.stringify({ status: "error", message: error.toString() }))
      .setMimeType(ContentService.MimeType.JSON);
  }
}
